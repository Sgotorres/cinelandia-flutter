-- =====================================================
-- 1. CREACIÓN DE LA BASE DE DATOS (Opcional si ya existe)
-- =====================================================
-- CREATE DATABASE restaurante_db;
-- \c restaurante_db;

-- Asegurar esquema público
CREATE SCHEMA IF NOT EXISTS public;

-- =====================================================
-- 2. TABLAS INDEPENDIENTES (Sin llaves foráneas salientes)
-- =====================================================

-- Tabla: configuracion_sistema
CREATE TABLE public.configuracion_sistema (
  clave character varying NOT NULL,
  valor jsonb NOT NULL,
  CONSTRAINT configuracion_sistema_pkey PRIMARY KEY (clave)
);

-- Tabla: usuarios
CREATE TABLE public.usuarios (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  username character varying UNIQUE,
  password_hash text,
  rol text,
  nombre character varying,
  apellido character varying,
  correo character varying,
  CONSTRAINT usuarios_pkey PRIMARY KEY (id)
);

-- Tabla: mesas
CREATE TABLE public.mesas (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  nombre character varying NOT NULL UNIQUE,
  activa boolean DEFAULT true,
  orden integer DEFAULT 0,
  CONSTRAINT mesas_pkey PRIMARY KEY (id)
);

-- Tabla: productos
CREATE TABLE public.productos (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  nombre character varying NOT NULL,
  descripcion text,
  precio numeric,
  categoria character varying,
  es_recomendado boolean DEFAULT false,
  disponible boolean DEFAULT true,
  precio_g numeric DEFAULT 0,
  precio_f numeric DEFAULT 0,
  precio_m numeric,
  volumen character varying,
  CONSTRAINT productos_pkey PRIMARY KEY (id)
);

-- Tabla: notificaciones
CREATE TABLE public.notificaciones (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  emisor character varying NOT NULL,
  receptor character varying NOT NULL,
  mensaje text NOT NULL,
  leido boolean DEFAULT false,
  fecha timestamp with time zone DEFAULT now(),
  CONSTRAINT notificaciones_pkey PRIMARY KEY (id)
);

-- =====================================================
-- 3. TABLAS DEPENDIENTES (Con llaves foráneas)
-- =====================================================

-- Tabla: pedidos (Depende de mesas)
CREATE TABLE public.pedidos (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  total numeric NOT NULL,
  estado character varying DEFAULT 'pendiente',
  fecha timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  mesa_id integer,
  CONSTRAINT pedidos_pkey PRIMARY KEY (id),
  CONSTRAINT pedidos_mesa_id_fkey FOREIGN KEY (mesa_id) 
    REFERENCES public.mesas(id) 
    ON DELETE SET NULL 
    ON UPDATE CASCADE
);

-- Tabla: detalles_pedido (Depende de pedidos y productos)
CREATE TABLE public.detalles_pedido (
  id integer GENERATED BY DEFAULT AS IDENTITY,
  pedido_id integer NOT NULL,
  producto_id integer NOT NULL,
  cantidad integer NOT NULL,
  precio_unitario numeric NOT NULL,
  talla character varying,
  producto_2_id integer,
  nota text,
  precio_extra numeric DEFAULT 0,
  CONSTRAINT detalles_pedido_pkey PRIMARY KEY (id),
  CONSTRAINT detalles_pedido_pedido_id_fkey FOREIGN KEY (pedido_id) 
    REFERENCES public.pedidos(id) 
    ON DELETE CASCADE 
    ON UPDATE CASCADE,
  CONSTRAINT detalles_pedido_producto_id_fkey FOREIGN KEY (producto_id) 
    REFERENCES public.productos(id) 
    ON DELETE RESTRICT 
    ON UPDATE CASCADE,
  CONSTRAINT detalles_pedido_producto_2_id_fkey FOREIGN KEY (producto_2_id) 
    REFERENCES public.productos(id) 
    ON DELETE SET NULL 
    ON UPDATE CASCADE
);

-- =====================================================
-- 4. ÍNDICES DE RENDIMIENTO (Claves foráneas frecuentes)
-- =====================================================
CREATE INDEX idx_pedidos_mesa_id ON public.pedidos(mesa_id);
CREATE INDEX idx_detalles_pedido_pedido_id ON public.detalles_pedido(pedido_id);
CREATE INDEX idx_detalles_pedido_producto_id ON public.detalles_pedido(producto_id);
CREATE INDEX idx_detalles_pedido_producto_2_id ON public.detalles_pedido(producto_2_id);
CREATE INDEX idx_notificaciones_receptor ON public.notificaciones(receptor);
